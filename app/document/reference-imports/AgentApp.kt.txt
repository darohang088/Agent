package com.daro.agentapp

import android.annotation.SuppressLint
import android.app.Application
import android.content.Context
import com.daro.agentapp.core.utils.EncryptedSharedPreferences
import com.daro.agentapp.core.utils.LanguageUtils
import java.util.Locale

class AgentApp : Application() {

    override fun attachBaseContext(newBase: Context) {
        val langCode = LanguageUtils.getLanguage(newBase)
        val context = LanguageUtils.applyLanguage(newBase, langCode)
        super.attachBaseContext(context)
    }

    override fun onCreate() {
        super.onCreate()
        instance = this
//        RATManager.init(this)
        EncryptedSharedPreferences.init(this)
    }

    companion object {
        @SuppressLint("StaticFieldLeak")
        @Volatile
        private var instance: AgentApp? = null

        /**
         * Expose as global access
         */
        val context  by lazy { instance ?: throw Exception("Context is not initialize") }
    }
}
